# ==============================================================================
# Dependencies for nl2sql-selected-schema-llamaindex
# Covers selected-schema-query.py and selected-schema-chat.py
# Python 3.10 or newer is recommended.
# Versions are left unpinned. If pip reports a resolver conflict, install into a
# clean virtual environment with a single "pip install -r requirements.txt".
# ==============================================================================

# --- LlamaIndex core (Settings, SQLDatabase, VectorStoreIndex, ObjectIndex,
#     SQLTableRetrieverQueryEngine, CondenseQuestionChatEngine, Memory, callbacks)
# The chat script uses llama_index.core.memory.Memory, so keep this package recent.
llama-index-core

# --- LlamaIndex integrations imported directly by the scripts
llama-index-vector-stores-chroma
llama-index-embeddings-google-genai
llama-index-llms-litellm

# --- Vector database client (chromadb.CloudClient for Chroma Cloud)
chromadb

# --- LLM routing and cost tracking (Gemini calls, fallbacks, completion_cost)
litellm

# --- Token counting (cl100k_base encoder used by TokenCountingHandler)
tiktoken

# --- Tracing with Arize Phoenix (phoenix.otel.register and the LlamaIndex instrumentor)
arize-phoenix-otel
openinference-instrumentation-llama-index

# --- SQL database access (create_engine, inspect, SQLDatabase)
sqlalchemy
# Synchronous PostgreSQL driver for DATABASE_URL (postgresql://...)
psycopg2-binary
# Async PostgreSQL driver for the chat memory (ASYNCPG_DATABASE_URL)
# Needed by selected-schema-chat.py only. The query script does not use it.
asyncpg

# --- Utilities
# Loads the .env file
python-dotenv
# Allows asyncio.run() to work inside already running event loops (CLI and Jupyter)
nest-asyncio
